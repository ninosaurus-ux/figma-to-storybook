# Colors

Token warna dari Figma, disync via Tokens Studio.
Ubah di Figma → Push → halaman ini ikut berubah otomatis.

export function collectColors() {
  const out = [];
  const walk = (node, path) => {
    for (const [key, val] of Object.entries(node)) {
      if (key.startsWith('$')) continue;
      const hasValue = val && typeof val === 'object' && ('value' in val || '$value' in val);
      if (hasValue) {
        const type = '$type' in val ? val.$type : val.type;
        if (type === 'color') out.push({ name: [...path, key].join('.'), token: val });
      } else if (val && typeof val === 'object') {
        walk(val, [...path, key]);
      }
    }
  };
  for (const [setName, set] of Object.entries(tokens)) {
    if (setName.startsWith('$')) continue;
    walk(set, []);
  }
  return out;
}

export const colors = collectColors();
export const varName = (name) => '--' + name.split('.').join('-').toLowerCase().replace(/[^a-z0-9-]/g, '');
export const desc = (token) => ('$description' in token ? token.$description : token.description);

  ))}
